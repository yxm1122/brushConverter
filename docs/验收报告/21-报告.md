## 三、关键数据（全部附取数命令）

# 三个核心数字
python -c "import json;d=json.load(open('docs/数据/ps_keys.json',encoding='utf-8'));print('len=',len(d));print('with_offset=',sum(1 for i in d if i['offset']));print('with_range=',sum(1 for i in d if i['range']))"

# 形态拆分
python -c "import json;d=json.load(open('docs/数据/ps_keys.json',encoding='utf-8'));print(sum(1 for i in d if i['form']=='string'), sum(1 for i in d if i['form']=='4cc_immediate'))"
→ 93 32

# 交叉核对
python -c "import json;ak=json.load(open('docs/数据/abr_keys.json',encoding='utf-8'));d=json.load(open('docs/数据/ps_keys.json',encoding='utf-8'));ks={i['key'] for i in d};inr=[k['key'] for k in ak['keys'] if k.get('in_repo_parser')];print('仓库有逆向未见',sorted(set(inr)-ks));print('逆向有仓库未见',len(ks-set(inr)))"
→ ['InvT','Txtr','Wtdg','sampledData'] / 90

### 3.1 47 行四字段校验脚本（`check_103`，未新增 tools 文件）

```python
# check_103：47 行四字段 + 枚举 + note 关键词 + 统计 + verdict 只读校验
import json, re, subprocess
from collections import Counter
d = json.load(open('docs/数据/mode-table.json', encoding='utf-8'))
old = json.loads(subprocess.check_output(['git', 'show', 'HEAD:docs/数据/mode-table.json']).decode('utf-8'))
rows, old_rows = d['modes'], {m['order']: m for m in old['modes']}
TYPES = {'像素实测', '源码', '逆向静态', '逆向动态', '文档', '推断'}
UI = {'Mltp', 'Sbtr', 'Drkn', 'Ovrl', 'CDdg', 'CBrn', 'linearBurn', 'hardMix', 'linearHeight', 'height'}
assert len(rows) == 47 and [m['order'] for m in rows] == list(range(47))
for m in rows:
    o = m['order']
    assert m['evidence_grade'] in {'A', 'B', 'C'}, o
    assert m['evidence_types'] and set(m['evidence_types']) <= TYPES, o
    assert re.fullmatch(r'\d{4}-\d{2}-\d{2}', m['recheck_at']), o
    assert m['recheck_cmd'], o
    assert m['evidence_strength_legacy'] == old_rows[o]['evidence_strength'], o
    assert m['verdict'] == old_rows[o]['verdict'], o
    assert m['evidence_grade'] != 'B' or '依赖' in m['note'], o
    assert m['evidence_grade'] != 'C' or 'PS UI 无此模式' in m['note'], o
    if m['recheck_cmd'] != '暂无（待任务 104）':
        assert not re.search(r'[A-Za-z]:[\\/]', m['recheck_cmd']), o
        for p in re.findall(r"'([^']+\.json)'", m['recheck_cmd']):
            assert __import__('pathlib').Path(p).exists(), (o, p)
g = dict(sorted(Counter(m['evidence_grade'] for m in rows).items()))
t = {x: sum(1 for m in rows if x in m['evidence_types']) for x in ['像素实测','源码','逆向静态','逆向动态','文档','推断']}
u = {'有': sum(1 for m in rows if m['ps_member'] in UI), '无': sum(1 for m in rows if m['ps_member'] not in UI)}
assert d['counts']['by_evidence_grade'] == g
assert d['counts']['by_evidence_types'] == t
assert d['counts']['by_ui_visibility'] == u
assert all(m['verdict'] == old_rows[m['order']]['verdict'] for m in rows)
print('PASS 47 rows | grade', g, '| types', t, '| ui', u)
print('PASS verdict 0 changed | legacy 47 kept | B 依赖', sum(1 for m in rows if m['evidence_grade']=='B'))
print('PASS recheck:', sum(1 for m in rows if m['recheck_cmd'].startswith('暂无')), '暂无 /', sum(1 for m in rows if not m['recheck_cmd'].startswith('暂无')), '可执行')
```
